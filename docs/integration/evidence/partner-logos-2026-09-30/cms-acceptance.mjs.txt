import {readFile,writeFile,mkdir} from 'node:fs/promises';
import {parseEnv} from 'node:util';
import {chromium,expect} from '@playwright/test';
const values=parseEnv(await readFile(process.argv[2],'utf8'));
if(new URL(values.DATABASE_URL_TEST).hostname!=='ep-plain-mouse-azm8pl2j-pooler.c-3.ap-southeast-1.aws.neon.tech'||!values.M2_TEST_STAFF_EMAIL?.endsWith('.example.test')||!values.M2_TEST_STAFF_PASSWORD)throw new Error('ISOLATED_CMS_IDENTITY_REQUIRED');
const state=JSON.parse(await readFile('.playwright/partner-r2-fixture-state.json','utf8'));

const browser=await chromium.launch({headless:true});

const evidence={environment:'isolated local',authentication:'existing Neon Auth email/password API with synthetic staff; no email send',anonymousProtected:false,providerSignInStatus:null,staffEditor:false,unpublishedPublicCount:null,republishedPublicCount:null,productionChanged:false};
try {
  const page=await context.newPage();
  const path='/zh/admin/partners/'+state.partnerIds[0];
  await page.goto(path,{timeout:120_000});await expect(page).toHaveURL(/\/zh\/admin-login/,{timeout:45_000});evidence.anonymousProtected=true;
  const response=await page.request.post('/api/auth/sign-in/email',{data:{email:values.M2_TEST_STAFF_EMAIL,password:values.M2_TEST_STAFF_PASSWORD,callbackURL:'/admin'},headers:{Origin:'http://localhost:3017'}});
  evidence.providerSignInStatus=response.status();expect(response.ok()).toBe(true);

} catch(error) {evidence.passed=false;evidence.error={name:error?.name};process.exitCode=1;}
finally {await writeFile('.playwright/partner-r2-cms-evidence.json',JSON.stringify(evidence,null,2)+'\n');console.log(JSON.stringify(evidence));await browser.close();}
