async function imagesReady(page,selector,count) {
  const images=page.locator(selector);
  await expect(images).toHaveCount(count);
  const bounds=await images.evaluateAll(images=>({first:images[0].getBoundingClientRect().top+window.scrollY,last:images.at(-1).getBoundingClientRect().bottom+window.scrollY}));
  for(let y=Math.max(0,bounds.first-150);y<bounds.last;y+=500) { await page.evaluate(y=>window.scrollTo({top:y,behavior:'instant'}),y); await page.waitForTimeout(80); }
  await page.waitForFunction(({selector,count})=>{const images=[...document.querySelectorAll(selector)];return images.length===count&&images.every(img=>img.complete&&img.naturalWidth>0&&img.naturalHeight>0)}, {selector,count}, {timeout:45_000});
}
